# 徒歩15分 駅マップ (walk15) 実装計画

> 開発時の計画の記録です。作業ツールやAIエージェントの指定はありません。現在の実装・起動手順は README.md、計算方法は docs/method.md を参照してください。チェックボックスは当時の計画であり、現在の未完了作業を表すものではありません。

**Goal:** 地点を選ぶと、歩く道のりで 15 分以内の駅と徒歩分数を示す静的な地図アプリ (東京都、島しょ部を除く) を作る。

**Architecture:** Python (uv) で道路網 (Overture) と駅 (N02) からグラフを作り、駅ごとに 1,200m で打ち切った最短路を求め、H3 解像度 11 のセルごとの「駅と道のり」の表・駅数の塗り・駅ごとの輪郭を静的ファイルに書き出す。画面は MapLibre GL JS と h3-js をビルドなしで使い、クリック地点のセルの表を引くだけで答えを出す。

**Tech Stack:** Python 3.13 / uv, duckdb, geopandas, shapely 2, pyproj, scipy (csgraph, cKDTree), h3 4, pyarrow, pytest。画面は MapLibre GL JS 6.11.2 (ESM)、h3-js 4.5.0 (ESM)、OpenFreeMap Liberty、地理院 シームレス空中写真。

設計書: `docs/superpowers/specs/2026-10-03-walk15-design.md`

```python
"""Walking time convention: 80 m per minute, fractions rounded up (minimum 1 minute).

This follows the Japanese real-estate advertising rule (不動産の表示に関する公正競争規約施行規則).
"""

import math

M_PER_MIN = 80.0
_EPS = 1e-9


def walk_minutes(distance_m: float, m_per_min: float = M_PER_MIN) -> int:
    return max(1, math.ceil(distance_m / m_per_min - _EPS))


def ceil_m(distance_m: float) -> int:
    """Distance stored in the app data: whole metres, rounded up."""
    return math.ceil(distance_m - _EPS)
```

```python
"""Download a file once, with a provenance record next to it."""

from __future__ import annotations

import hashlib
import json
import urllib.request
from datetime import datetime, timezone
from pathlib import Path

USER_AGENT = "eki-walk (personal open-data project)"


def download(url: str, dest: Path | str, expected_size: int | None = None) -> dict:
    dest = Path(dest)
    dest.parent.mkdir(parents=True, exist_ok=True)
    part = dest.with_name(dest.name + ".part")
    req = urllib.request.Request(url, headers={"User-Agent": USER_AGENT})
    h = hashlib.sha256()
    n = 0
    with urllib.request.urlopen(req) as r, open(part, "wb") as f:
        while chunk := r.read(1 << 20):
            f.write(chunk)
            h.update(chunk)
            n += len(chunk)
    if expected_size is not None and n != expected_size:
        part.unlink()
        raise ValueError(f"size mismatch for {url}: expected {expected_size}, got {n}")
    part.replace(dest)
    prov = {
        "url": url,
        "file": dest.name,
        "bytes": n,
        "sha256": h.hexdigest(),
        "retrieved_at_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
    }
    write_provenance(dest, prov)
    return prov


def write_provenance(path: Path | str, prov: dict) -> Path:
    path = Path(path)
    p = path.with_name(path.name + ".provenance.json")
    p.write_text(json.dumps(prov, ensure_ascii=False, indent=2), encoding="utf-8")
    return p


def read_provenance(path: Path | str) -> dict:
    path = Path(path)
    return json.loads(path.with_name(path.name + ".provenance.json").read_text(encoding="utf-8"))


def sha256_file(path: Path | str) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(1 << 20):
            h.update(chunk)
    return h.hexdigest()
```

```python
"""Walkable road network from Overture transportation segments.

Nodes are Overture connectors; edges are the pieces of each segment between
consecutive connectors. Lengths are planar metres (the GeoDataFrame must already
be in a metric CRS). The graph is undirected.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import pandas as pd
import shapely
from scipy.sparse import coo_matrix
from scipy.sparse.csgraph import connected_components
from shapely.geometry import LineString
from shapely.ops import substring

EXCLUDED_CLASSES = {"motorway"}
CONDITIONAL_KEYS = ("during", "heading", "using", "recognized", "vehicle")


def _as_list(x) -> list:
    if x is None:
        return []
    if isinstance(x, float) and np.isnan(x):
        return []
    return list(x)


def _nonempty(x) -> bool:
    """Truthiness that also works for numpy arrays coming from pyarrow."""
    if x is None:
        return False
    if isinstance(x, str):
        return bool(x)
    if hasattr(x, "__len__"):
        return len(x) > 0
    return bool(x)


def foot_allowed(rules) -> bool:
    """True unless walking is denied unconditionally. An explicit foot allow beats a blanket deny."""
    allowed = False
    denied = False
    for r in _as_list(rules):
        when = r.get("when") or {}
        if any(_nonempty(when.get(k)) for k in CONDITIONAL_KEYS):
            continue
        modes = _as_list(when.get("mode"))
        if modes and "foot" not in modes:
            continue
        kind = r.get("access_type")
        if kind in ("allowed", "designated") and "foot" in modes:
            allowed = True
        elif kind == "denied":
            denied = True
    return allowed or not denied


def under_construction(flags) -> bool:
    for f in _as_list(flags):
        if f.get("between") is None and "is_under_construction" in _as_list(f.get("values")):
            return True
    return False


def walkable_mask(df: pd.DataFrame) -> pd.Series:
    road = df["subtype"].eq("road")
    cls_ok = ~df["class"].isin(EXCLUDED_CLASSES)
    foot = df["access_restrictions"].map(foot_allowed)
    built = ~df["road_flags"].map(under_construction)
    return road & cls_ok & foot & built


def split_segment(geom: LineString, connectors) -> list[tuple[str, str, LineString]]:
    cs = sorted(((float(c["at"]), c["connector_id"]) for c in _as_list(connectors)), key=lambda x: x[0])
    out = []
    for (a0, c0), (a1, c1) in zip(cs, cs[1:]):
        if c0 == c1:
            continue
        if a0 <= 0.0 and a1 >= 1.0:
            piece = geom
        else:
            piece = substring(geom, a0, a1, normalized=True)
        if piece.geom_type != "LineString":  # zero-length piece
            p = piece.coords[0]
            piece = LineString([p, p])
        out.append((c0, c1, piece))
    return out


@dataclass
class Graph:
    node_ids: np.ndarray  # str, shape (N,)
    node_xy: np.ndarray  # float, shape (N, 2)
    edge_u: np.ndarray  # int, shape (E,)
    edge_v: np.ndarray  # int, shape (E,)
    edge_len: np.ndarray  # float metres, shape (E,)
    edge_geom: np.ndarray  # shapely LineString (metric), shape (E,)


def build_graph(segments) -> Graph:
    index: dict[str, int] = {}
    xy: list[tuple[float, float]] = []
    us, vs, geoms = [], [], []

    def node(cid: str, p) -> int:
        i = index.get(cid)
        if i is None:
            i = index[cid] = len(xy)
            xy.append((p[0], p[1]))
        return i

    for geom, conns in zip(segments.geometry.values, segments["connectors"].values):
        for c0, c1, piece in split_segment(geom, conns):
            us.append(node(c0, piece.coords[0]))
            vs.append(node(c1, piece.coords[-1]))
            geoms.append(piece)

    ids = np.empty(len(index), dtype=object)
    for cid, i in index.items():
        ids[i] = cid
    geoms_arr = np.array(geoms, dtype=object)
    return Graph(
        node_ids=ids,
        node_xy=np.array(xy, dtype=float).reshape(-1, 2),
        edge_u=np.array(us, dtype=np.int64),
        edge_v=np.array(vs, dtype=np.int64),
        edge_len=shapely.length(geoms_arr).astype(float),
        edge_geom=geoms_arr,
    )


def largest_component(g: Graph) -> Graph:
    n = len(g.node_ids)
    adj = coo_matrix((np.ones(len(g.edge_u)), (g.edge_u, g.edge_v)), shape=(n, n))
    _, labels = connected_components(adj, directed=False)
    keep_label = np.bincount(labels).argmax()
    keep = labels == keep_label
    new_index = np.full(n, -1, dtype=np.int64)
    new_index[keep] = np.arange(keep.sum())
    e = keep[g.edge_u] & keep[g.edge_v]
    return Graph(
        node_ids=g.node_ids[keep],
        node_xy=g.node_xy[keep],
        edge_u=new_index[g.edge_u[e]],
        edge_v=new_index[g.edge_v[e]],
        edge_len=g.edge_len[e],
        edge_geom=g.edge_geom[e],
    )
```

```python
"""Shortest walking distance from stations, cut off at a limit."""

from __future__ import annotations

import numpy as np
import pandas as pd
from scipy.sparse import csr_matrix
from scipy.sparse.csgraph import dijkstra
from scipy.spatial import cKDTree

from eki_walk.network.graph import Graph

MIN_WEIGHT = 0.01  # csgraph drops zero-weight entries; keep a tiny positive weight


def seed_nodes(node_xy: np.ndarray, point_xy: np.ndarray, radius: float, tree: cKDTree | None = None):
    """Nodes within `radius` of the point (offset = straight distance); else the nearest node."""
    tree = tree or cKDTree(node_xy)
    idx = np.asarray(tree.query_ball_point(point_xy, r=radius), dtype=np.int64)
    if idx.size == 0:
        d, i = tree.query(point_xy)
        return np.array([i], dtype=np.int64), np.array([float(d)])
    off = np.hypot(*(node_xy[idx] - point_xy).T)
    return idx, off


def _dedupe_min(r, c, w):
    order = np.lexsort((w, c, r))
    r, c, w = r[order], c[order], w[order]
    keep = np.ones(len(r), dtype=bool)
    keep[1:] = (r[1:] != r[:-1]) | (c[1:] != c[:-1])
    return r[keep], c[keep], w[keep]


def reach_from_points(
    g: Graph, points_xy: np.ndarray, seed_radius: float, limit: float, batch: int = 8
) -> pd.DataFrame:
    """Return rows (station, node, dist) for every node within `limit` of each station point."""
    n = len(g.node_ids)
    s = len(points_xy)
    tree = cKDTree(g.node_xy)
    loops = g.edge_u == g.edge_v
    eu, ev, el = g.edge_u[~loops], g.edge_v[~loops], g.edge_len[~loops]
    rows = [eu, ev]
    cols = [ev, eu]
    ws = [el, el]
    for i, p in enumerate(points_xy):
        idx, off = seed_nodes(g.node_xy, p, seed_radius, tree)
        rows.append(np.full(len(idx), n + i, dtype=np.int64))
        cols.append(idx)
        ws.append(off)
    r, c, w = _dedupe_min(np.concatenate(rows), np.concatenate(cols), np.concatenate(ws).astype(float))
    w = np.maximum(w, MIN_WEIGHT)
    m = csr_matrix((w, (r, c)), shape=(n + s, n + s))

    out = []
    for start in range(0, s, batch):
        ind = np.arange(n + start, n + min(s, start + batch))
        dist = dijkstra(m, directed=True, indices=ind, limit=limit)[:, :n]
        si, ni = np.nonzero(np.isfinite(dist))
        out.append(
            pd.DataFrame({"station": start + si, "node": ni, "dist": dist[si, ni]})
        )
    if not out:
        return pd.DataFrame({"station": [], "node": [], "dist": []})
    return pd.concat(out, ignore_index=True)
```

注: 起点のオフセットは最小 `MIN_WEIGHT` (0.01m)。駅の中心が節点の真上でも距離は 0.01m になる。

```python
"""Stations from N02: selection, platform centres, and grouping of same-name stations."""

from __future__ import annotations

import re

import geopandas as gpd
import numpy as np
import pandas as pd
import shapely


def select_stations(st: gpd.GeoDataFrame, aoi, buffer_m: float, exclude_classes) -> gpd.GeoDataFrame:
    near = st.geometry.intersects(aoi.buffer(buffer_m))
    ok = ~st["rail_class"].isin(list(exclude_classes))
    return st.loc[near & ok].reset_index(drop=True)


def _main_line(g):
    g = shapely.line_merge(g) if g.geom_type == "MultiLineString" else g
    if g.geom_type == "MultiLineString":
        g = max(g.geoms, key=lambda p: p.length)
    return g


def platform_centers(geoms) -> np.ndarray:
    """Midpoint along each platform line (the longest part if the line is broken)."""
    lines = np.array([_main_line(g) for g in geoms], dtype=object)
    pts = shapely.line_interpolate_point(lines, 0.5, normalized=True)
    return shapely.get_coordinates(pts)


def station_groups(st: pd.DataFrame) -> pd.DataFrame:
    """One row per N02 group code. Name = name of the representative station (code == group code)."""
    rows = []
    for code, part in st.groupby("group_code", sort=True):
        rep = part[part["station_code"] == code]
        name = (rep if len(rep) else part)["name"].iloc[0]
        rows.append(
            {
                "group_code": code,
                "name": name,
                "cx": float(part["cx"].mean()),
                "cy": float(part["cy"].mean()),
                "members": part["station_code"].tolist(),
            }
        )
    return pd.DataFrame(rows)


_NUMBERED = re.compile(r"^\d+号線(?=.)")


def display_line(line: str) -> str:
    """'4号線丸ノ内線' -> '丸ノ内線' (N02 prefixes some subway lines with their number)."""
    return _NUMBERED.sub("", line)
```

```python
from __future__ import annotations

import zipfile
from pathlib import Path

import geopandas as gpd
import shapely


def _member(zip_path: Path, suffixes: tuple[str, ...]) -> str:
    with zipfile.ZipFile(zip_path) as z:
        names = [n for n in z.namelist() if n.lower().endswith(suffixes)]
    if not names:
        raise FileNotFoundError(f"no {suffixes} in {zip_path}")
    return sorted(names, key=len)[0]


def read_zip_layer(zip_path: Path | str, suffixes=(".geojson", ".shp")) -> gpd.GeoDataFrame:
    zip_path = Path(zip_path)
    for suffix in suffixes:
        try:
            inner = _member(zip_path, (suffix,))
        except FileNotFoundError:
            continue
        return gpd.read_file(f"zip://{zip_path.as_posix()}!{inner}")
    raise FileNotFoundError(f"no {suffixes} in {zip_path}")

    keep = ~gdf["N03_002"].isin(list(exclude_n03_002))
    if only_n03_004:
        keep &= gdf["N03_004"].isin(list(only_n03_004))
    return shapely.union_all(shapely.make_valid(gdf.loc[keep].geometry.values))
```

```python
"""H3 cells, snapping of cell centres to the walkable network, and per-cell distances."""

from __future__ import annotations

import h3
import numpy as np
import pandas as pd
import shapely
from pyproj import Transformer

from eki_walk.network.graph import Graph


def aoi_cells(aoi_wgs84, res: int) -> np.ndarray:
    """Cells whose centre lies in the AOI (lon/lat geometry)."""
    return np.array(sorted(h3.geo_to_cells(aoi_wgs84, res)), dtype=object)


def cell_centers_xy(cells, metric_crs: str) -> np.ndarray:
    latlng = np.array([h3.cell_to_latlng(c) for c in cells], dtype=float).reshape(-1, 2)
    tr = Transformer.from_crs("EPSG:4326", metric_crs, always_xy=True)
    x, y = tr.transform(latlng[:, 1], latlng[:, 0])
    return np.column_stack([x, y])


def snap_points(points_xy: np.ndarray, edge_geoms: np.ndarray, max_dist: float) -> pd.DataFrame:
    """Nearest edge within max_dist for each point: rows (point, edge, snap_d, t)."""
    tree = shapely.STRtree(edge_geoms)
    pts = shapely.points(points_xy)
    (pi, ei), d = tree.query_nearest(pts, max_distance=max_dist, return_distance=True, all_matches=False)
    t = shapely.line_locate_point(edge_geoms[ei], pts[pi])
    return pd.DataFrame({"point": pi, "edge": ei, "snap_d": d, "t": t})


def cell_line_distances(snap: pd.DataFrame, g: Graph, reach: pd.DataFrame, limit: float) -> pd.DataFrame:
    """Rows (point, station, dist): walk from the snapped point to each station via either edge end."""
    e = snap["edge"].to_numpy()
    length = g.edge_len[e]
    t = np.clip(snap["t"].to_numpy(), 0, length)
    s = snap["snap_d"].to_numpy()
    p = snap["point"].to_numpy()
    ends = pd.concat(
        [
            pd.DataFrame({"point": p, "node": g.edge_u[e], "add": s + t}),
            pd.DataFrame({"point": p, "node": g.edge_v[e], "add": s + (length - t)}),
        ],
        ignore_index=True,
    )
    m = ends.merge(reach, on="node", how="inner")
    m["dist"] = m["dist"] + m["add"]
    out = m.groupby(["point", "station"], as_index=False)["dist"].min()
    return out[out["dist"] <= limit].reset_index(drop=True)
```

```python
"""Write the static files that the walk15 web app reads."""

from __future__ import annotations

import json
from collections import defaultdict
from pathlib import Path

import h3
import numpy as np
import pandas as pd
import shapely
from pyproj import Transformer
from shapely.geometry import mapping, shape

from eki_walk.units import ceil_m

SIMPLIFY_M = 3.0
PRECISION_DEG = 1e-6


def cells_to_geometry(cells):
    """Union of H3 cells as a shapely (Multi)Polygon in lon/lat."""
    return shape(h3.cells_to_h3shape(list(cells)).__geo_interface__)


def simplify_lonlat(geom, metric_crs: str):
    fwd = Transformer.from_crs("EPSG:4326", metric_crs, always_xy=True)
    inv = Transformer.from_crs(metric_crs, "EPSG:4326", always_xy=True)
    g = shapely.transform(geom, lambda xy: np.column_stack(fwd.transform(xy[:, 0], xy[:, 1])))
    g = g.simplify(SIMPLIFY_M, preserve_topology=True)
    g = shapely.transform(g, lambda xy: np.column_stack(inv.transform(xy[:, 0], xy[:, 1])))
    return shapely.set_precision(g, PRECISION_DEG)


def feature(geom, props) -> dict:
    return {"type": "Feature", "properties": props, "geometry": mapping(geom)}


def write_json(path: Path, obj) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, ensure_ascii=False, separators=(",", ":")), encoding="utf-8")


def write_cell_chunks(cells: np.ndarray, table: pd.DataFrame, far_points: np.ndarray, chunk_res: int, out_dir: Path) -> int:
    """table rows: (point, line, dist). One JSON per parent cell at chunk_res."""
    chunks: dict[str, dict] = defaultdict(lambda: {"cells": {}, "far": []})
    t = table.sort_values(["point", "dist"])
    for point, part in t.groupby("point", sort=False):
        cell = cells[point]
        flat = []
        for line, dist in zip(part["line"].to_numpy(), part["dist"].to_numpy()):
            flat += [int(line), ceil_m(float(dist))]
        chunks[h3.cell_to_parent(cell, chunk_res)]["cells"][cell] = flat
    for point in far_points:
        cell = cells[point]
        chunks[h3.cell_to_parent(cell, chunk_res)]["far"].append(cell)
    res = h3.get_resolution(cells[0]) if len(cells) else None
    for parent, body in chunks.items():
        write_json(Path(out_dir) / "cells" / f"{parent}.json", {"res": res, **body})
    return len(chunks)


def count_features(cells: np.ndarray, group_counts: pd.Series, metric_crs: str, max_band: int = 5) -> list[dict]:
    """group_counts: index = point, value = number of station groups within the limit."""
    bands = group_counts.clip(upper=max_band)
    feats = []
    for band in sorted(bands.unique()):
        pts = bands.index[bands == band].to_numpy()
        geom = simplify_lonlat(cells_to_geometry(cells[pts]), metric_crs)
        feats.append(feature(geom, {"count": int(band)}))
    return feats


def isochrone_features(cells: np.ndarray, group_table: pd.DataFrame, metric_crs: str) -> list[dict]:
    """group_table rows: (point, group, dist) already limited to the walking limit."""
    feats = []
    for group, part in group_table.groupby("group", sort=True):
        geom = simplify_lonlat(cells_to_geometry(cells[part["point"].to_numpy()]), metric_crs)
        feats.append(feature(geom, {"group": int(group)}))
    return feats


def feature_collection(features: list[dict]) -> dict:
    return {"type": "FeatureCollection", "features": features}
```

```python
"""Extract Overture transportation segments for a bounding box with DuckDB."""

from __future__ import annotations

from pathlib import Path

import duckdb

BUCKET = "s3://overturemaps-us-west-2/release"
COLUMNS = ["id", "subtype", "class", "subclass", "connectors", "access_restrictions", "road_flags", "geometry"]


def segment_source(release: str) -> str:
    return f"{BUCKET}/{release}/theme=transportation/type=segment/*.parquet"


def _bbox_where(prefix: str, bbox) -> str:
    xmin, ymin, xmax, ymax = bbox
    return (
        f"{prefix}xmin <= {xmax} AND {prefix}xmax >= {xmin} "
        f"AND {prefix}ymin <= {ymax} AND {prefix}ymax >= {ymin}"
    )


def segment_sql(release: str, bbox, out_path: str) -> str:
    cols = ", ".join(f'"{c}"' for c in COLUMNS)
    return (
        f"COPY (SELECT {cols} FROM read_parquet('{segment_source(release)}') "
        f"WHERE subtype = 'road' AND {_bbox_where('bbox.', bbox)}) "
        f"TO '{out_path}' (FORMAT parquet, COMPRESSION zstd)"
    )


def estimate_sql(release: str, bbox) -> str:
    """Row groups whose bbox statistics intersect the bbox (reads Parquet footers only)."""
    src = segment_source(release)
    xmin, ymin, xmax, ymax = bbox
    return f"""
WITH m AS (SELECT * FROM parquet_metadata('{src}')),
rg AS (
  SELECT file_name, row_group_id,
         any_value(row_group_num_rows) AS nrows,
         sum(total_compressed_size) AS bytes,
         min(CASE WHEN path_in_schema = 'bbox, xmin' THEN TRY_CAST(stats_min AS DOUBLE) END) AS xmin,
         max(CASE WHEN path_in_schema = 'bbox, xmax' THEN TRY_CAST(stats_max AS DOUBLE) END) AS xmax,
         min(CASE WHEN path_in_schema = 'bbox, ymin' THEN TRY_CAST(stats_min AS DOUBLE) END) AS ymin,
         max(CASE WHEN path_in_schema = 'bbox, ymax' THEN TRY_CAST(stats_max AS DOUBLE) END) AS ymax
  FROM m GROUP BY file_name, row_group_id)
SELECT count(*) AS row_groups, count(DISTINCT file_name) AS files,
       sum(nrows) AS rows, sum(bytes) AS compressed_bytes
FROM rg WHERE xmin <= {xmax} AND xmax >= {xmin} AND ymin <= {ymax} AND ymax >= {ymin}
"""


def connect() -> duckdb.DuckDBPyConnection:
    con = duckdb.connect()
    con.execute("INSTALL httpfs; LOAD httpfs; SET s3_region = 'us-west-2';")
    return con


def estimate(release: str, bbox) -> dict:
    con = connect()
    row = con.execute(estimate_sql(release, bbox)).fetchone()
    return dict(zip(["row_groups", "files", "rows", "compressed_bytes"], row))


def fetch_segments(release: str, bbox, out_path: Path) -> None:
    out_path.parent.mkdir(parents=True, exist_ok=True)
    tmp = out_path.with_name(out_path.name + ".part")
    con = connect()
    con.execute(segment_sql(release, bbox, tmp.as_posix()))
    tmp.replace(out_path)
```

```python
        "bytes": out.stat().st_size,
        "sha256": sha256_file(out),

    }
```

```python
"""Command line entry point: eki-walk <command> --config configs/tokyo.toml"""

from __future__ import annotations

import argparse
import json

from eki_walk import pipeline
from eki_walk.config import load_config
from eki_walk.sources.download import download


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="eki-walk")
    sub = p.add_subparsers(dest="command", required=True)

    def add(name: str, **kw):
        sp = sub.add_parser(name, **kw)
        sp.add_argument("--config", required=True)
        return sp

    add("show-sources", help="print planned downloads (name, URL, size)")
    f = add("fetch", help="download a 国土数値情報 file")
    f.add_argument("source", choices=["n02", "n03"])
    add("build-aoi", help="build the area of interest and network bbox")
    add("overture-estimate", help="estimate the Overture read (Parquet metadata only)")
    add("fetch-overture", help="extract Overture road segments for the network bbox")
    b = add("build", help="run stations -> network -> reach -> walk15")
    b.add_argument("--from", dest="start", choices=pipeline.STEPS, default="stations")
    return p


def main(argv=None) -> None:
    a = build_parser().parse_args(argv)
    c = load_config(a.config)
    if a.command == "show-sources":
        for s in (c.n02, c.n03):
            print(f"{s.name}\t{s.url}\t{s.size} bytes")
        print(f"overture segment {c.overture_release}\t{pipeline.overture.segment_source(c.overture_release)}")
    elif a.command == "fetch":
        s = getattr(c, a.source)
        print(json.dumps(download(s.url, pipeline.ksj_path(c, a.source), s.size), indent=2))
    elif a.command == "build-aoi":
        print(json.dumps(pipeline.build_aoi(c), indent=2))
    elif a.command == "overture-estimate":
        est = pipeline.overture.estimate(c.overture_release, pipeline.network_bbox(c))
        print(json.dumps(est, indent=2, default=int))
    elif a.command == "fetch-overture":
        print(json.dumps(pipeline.fetch_overture(c), indent=2))
    elif a.command == "build":
        pipeline.run_build(c, a.start)


if __name__ == "__main__":
    main()
```

```js
  const r = await fetch(DATA + path);
  if (!r.ok) throw new Error(`${path}: ${r.status}`);
  return r.json();
}

// --- AOI point-in-polygon (rings in lon/lat) ---
function inRing(ring, x, y) {
  let inside = false;
  for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
    const [xi, yi] = ring[i];
    const [xj, yj] = ring[j];
    if ((yi > y) !== (yj > y) && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi) inside = !inside;
  }
  return inside;
}
function inAoi(lng, lat) {
  for (const f of state.aoi.features) {
    const g = f.geometry;
    const polys = g.type === "Polygon" ? [g.coordinates] : g.coordinates;
    for (const [outer, ...holes] of polys) {
      if (inRing(outer, lng, lat) && !holes.some((h) => inRing(h, lng, lat))) return true;
    }
  }
  return false;
}

async function loadChunk(parent) {
  if (!state.chunks.has(parent)) {
    state.chunks.set(
      parent,
      fetch(`${DATA}cells/${parent}.json`).then((r) => (r.ok ? r.json() : { cells: {}, far: [] })),
    );
  }
  return state.chunks.get(parent);
}

async function lookup(lng, lat) {
  if (!inAoi(lng, lat)) return { status: "outside" };
  const cell = h3.latLngToCell(lat, lng, state.meta.h3_res);
  const chunk = await loadChunk(h3.cellToParent(cell, state.meta.chunk_res));
  if (chunk.far.includes(cell)) return { status: "far" };
  const flat = chunk.cells[cell] || [];
  const byGroup = new Map();
  for (let i = 0; i < flat.length; i += 2) {
    const line = state.stations.lines[flat[i]];
    const d = flat[i + 1];
    const g = byGroup.get(line.group) || { group: line.group, dist: Infinity, lines: [] };
    g.dist = Math.min(g.dist, d);
    g.lines.push({ ...line, dist: d });
    byGroup.set(line.group, g);
  }
  const groups = [...byGroup.values()].sort((a, b) => a.dist - b.dist);
  return { status: "ok", groups };
}

function el(tag, attrs = {}, ...children) {
  const e = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (k === "class") e.className = v;
    else if (k.startsWith("on")) e.addEventListener(k.slice(2), v);
    else e.setAttribute(k, v);
  }
  for (const c of children) e.append(c);
  return e;
}

  const list = el("ol", { class: "stations" });
  for (const g of res.groups) {
    const info = state.stations.groups[g.group];
    const lines = g.lines
      .sort((a, b) => a.dist - b.dist)
      .map((l) => `${l.line} ${minutes(l.dist)}分`)
      .join(" · ");
    const item = el(

  showListed(res.groups.map((g) => g.group));

}
```
